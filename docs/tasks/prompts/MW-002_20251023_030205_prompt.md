### Task Description
Develop a GoFlow middleware for role-based access control. This middleware will leverage the authenticated user's roles, obtained from the context by `MW-001`, to determine access permissions for requested resources. It will verify if the user's roles satisfy the requirements for the target endpoint. Unauthorized access attempts will result in an `http.StatusForbidden` response, ensuring secure resource access.

### Acceptance Criteria
1. Middleware retrieves user roles from `context.Context`.
2. Denies requests with `http.StatusForbidden` if user lacks required role.
3. Permits requests if user possesses at least one required role.
4. Required roles are declaratively defined for protected endpoints.
5. Returns a standardized, clear error message for unauthorized access.

func (r *WorkflowRepository) FindByID(ctx context.Context, id string) (*domain.Workflow, error) {
    if id == "" {
        return nil, fmt.Errorf("workflow ID cannot be empty")
    }

    query := `SELECT id, name, version, definition, created_at, updated_at
              FROM workflows WHERE id = $1 AND deleted_at IS NULL`

    var workflow domain.Workflow
    err := r.db.QueryRowContext(ctx, query, id).Scan(
        &workflow.ID,
        &workflow.Name,
        &workflow.Version,
        &workflow.Definition,
        &workflow.CreatedAt,
        &workflow.UpdatedAt,
    )

    if err == sql.ErrNoRows {
        return nil, &NotFoundError{Resource: "workflow", ID: id}
    }
    if err != nil {
        return nil, fmt.Errorf("failed to query workflow: %w", err)
    }

    return &workflow, nil
}
